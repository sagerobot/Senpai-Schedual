  const renderSeriesGroup = (group: typeof queueData.queue) => {
    const firstItem = group[0];
    const seriesTitle = firstItem.anime.title.english || firstItem.anime.title.userPreferred;
    
    const totalWatched = group.reduce((acc, item) => acc + item.watched.length, 0);
    const totalAired = group.reduce((acc, item) => acc + item.airedCount, 0);
    const totalProgress = totalAired > 0 ? (totalWatched / totalAired) * 100 : 0;
    
    let currentSeasonIndex = -1;
    let nextEpisodeToWatch = -1;
    let currentItem: typeof queueData.queue[0] | null = null;
    
    for (let i = 0; i < group.length; i++) {
      const item = group[i];
      const unwatched = Array.from({ length: item.airedCount }, (_, idx) => idx + 1).find(ep => !item.watched.includes(ep));
      if (unwatched) {
        currentSeasonIndex = i;
        nextEpisodeToWatch = unwatched;
        currentItem = item;
        break;
      }
    }
    
    if (currentSeasonIndex === -1 && group.length > 0) {
       currentSeasonIndex = 0;
       currentItem = group[0];
    }
    
    const watchLink = currentItem?.anime.externalLinks?.find((l: any) => 
      ['Crunchyroll', 'Netflix', 'Hulu', 'Amazon Prime Video', 'HIDIVE', 'Disney Plus', 'Bilibili TV'].includes(l.site)
    )?.url || currentItem?.anime.externalLinks?.[0]?.url;

    return (
      <motion.div
        layout
        key={`group-${firstItem.anime.id}`}
        initial={{ opacity: 0, scale: 0.95 }}
        animate={{ opacity: 1, scale: 1 }}
        exit={{ opacity: 0, scale: 0.95 }}
        className="col-span-1 sm:col-span-2 md:col-span-2 lg:col-span-3 xl:col-span-3 2xl:col-span-4 flex flex-col rounded-2xl border border-[#2e267a] bg-[#0a0c16] shadow-xl relative overflow-hidden group/series"
      >
         <div className="p-4 sm:p-5 flex flex-col sm:flex-row gap-5 flex-1">
           {/* Left: Image */}
           <div className="relative shrink-0 sm:w-[140px] md:w-[180px] lg:w-[220px] cursor-pointer" onClick={() => onAnimeSelect(firstItem.anime)}>
             <img src={firstItem.anime.coverImage.large} className="w-full h-[240px] sm:h-full object-cover rounded-xl shadow-md" />
             <div className="absolute inset-0 rounded-xl shadow-[inset_0_0_20px_rgba(0,0,0,0.5)] pointer-events-none" />
             <div className="absolute bottom-3 left-3 right-3 text-center">
               <h3 className="font-bold text-white text-lg leading-tight line-clamp-3 drop-shadow-md">
                 {seriesTitle}
               </h3>
             </div>
           </div>
           
           {/* Right Content */}
           <div className="flex flex-1 flex-col min-w-0">
              <div className="flex justify-between items-start gap-4">
                <div className="min-w-0">
                  <div className="flex flex-wrap items-center gap-3 mb-2">
                    <h3 className="font-bold text-gray-100 text-xl md:text-2xl leading-tight truncate cursor-pointer hover:text-indigo-400 transition-colors" onClick={() => onAnimeSelect(firstItem.anime)}>
                      {seriesTitle}
                    </h3>
                    <span className="shrink-0 px-2.5 py-1 rounded-md bg-indigo-500/20 text-indigo-300 text-xs font-semibold border border-indigo-500/30">
                      {group.length} seasons
                    </span>
                  </div>
                  
                  {currentItem && nextEpisodeToWatch !== -1 && (
                    <div className="flex items-center gap-2 text-indigo-400 mb-4">
                      <Play className="w-4 h-4 fill-current shrink-0" />
                      <span className="font-semibold text-sm whitespace-nowrap">Continue in order</span>
                      <span className="text-gray-500 text-sm ml-1 truncate">Next up: Season {currentSeasonIndex + 1} • Episode {nextEpisodeToWatch}</span>
                    </div>
                  )}
                </div>
                
                {/* Progress Ring */}
                <div className="shrink-0 relative w-14 h-14 md:w-16 md:h-16 flex items-center justify-center">
                  <svg className="w-full h-full -rotate-90" viewBox="0 0 36 36">
                    <path
                      className="text-[#1e2336]"
                      d="M18 2.0845 a 15.9155 15.9155 0 0 1 0 31.831 a 15.9155 15.9155 0 0 1 0 -31.831"
                      fill="none"
                      stroke="currentColor"
                      strokeWidth="4"
                    />
                    <path
                      className="text-indigo-500 transition-all duration-1000 ease-out"
                      strokeDasharray={`${totalProgress}, 100`}
                      d="M18 2.0845 a 15.9155 15.9155 0 0 1 0 31.831 a 15.9155 15.9155 0 0 1 0 -31.831"
                      fill="none"
                      stroke="currentColor"
                      strokeWidth="4"
                      strokeLinecap="round"
                    />
                  </svg>
                  <div className="absolute flex flex-col items-center justify-center">
                    <span className="text-white font-bold text-sm md:text-base leading-none">{Math.round(totalProgress)}%</span>
                    <span className="text-[9px] text-gray-400 leading-none mt-0.5">watched</span>
                  </div>
                </div>
              </div>
              
              {/* Seasons List */}
              <div className="mt-4 flex flex-col gap-3 relative flex-1 min-h-0 overflow-y-auto pr-2 custom-scrollbar">
                {group.map((season, idx) => {
                   const isCurrent = idx === currentSeasonIndex;
                   const seasonProgress = season.airedCount > 0 ? (season.watched.length / season.airedCount) * 100 : 0;
                   return (
                     <div key={season.anime.id} className="relative flex items-center cursor-pointer group/row" onClick={() => onAnimeSelect(season.anime)}>
                       {/* The dotted line between circles */}
                       {idx < group.length - 1 && (
                         <div className="absolute left-4 top-[50%] h-[calc(100%+12px)] w-[2px] border-l-2 border-dashed border-[#2e267a] z-0" />
                       )}
                       
                       <div className={cn("relative z-10 w-8 h-8 rounded-full flex items-center justify-center text-xs font-bold border-2 shrink-0 transition-colors", 
                          isCurrent ? "bg-indigo-600 border-indigo-400 text-white shadow-[0_0_10px_rgba(99,102,241,0.4)]" : "bg-[#161b2e] border-[#2e267a] text-gray-400 group-hover/row:border-indigo-500/50")}>
                         {idx + 1}
                       </div>
                       
                       <div className={cn("ml-4 flex-1 flex flex-col justify-center rounded-xl border p-3 sm:px-4 transition-colors",
                          isCurrent ? "border-indigo-500/50 bg-[#121629]" : "border-[#1e2336] bg-[#0a0c16] group-hover/row:bg-[#121629]")}>
                          <div className="flex items-center justify-between gap-2">
                             <div className="flex items-baseline gap-2">
                               <span className={cn("font-bold text-[15px]", isCurrent ? "text-gray-100" : "text-gray-400 group-hover/row:text-gray-300")}>Season {idx + 1}</span>
                               <span className="text-xs font-medium text-gray-500">{season.watched.length}/{season.airedCount}</span>
                             </div>
                             
                             {isCurrent ? (
                               <span className="text-[11px] font-bold px-2.5 py-1 rounded-md bg-indigo-500/20 text-indigo-400 border border-indigo-500/20">Current</span>
                             ) : idx > currentSeasonIndex ? (
                               <span className="text-[11px] font-bold px-2.5 py-1 rounded-md bg-[#1e2336] text-gray-400 border border-[#2a3045]">Up Next</span>
                             ) : (
                               <span className="text-[11px] font-bold px-2.5 py-1 rounded-md bg-green-500/10 text-green-500/80 border border-green-500/20">Done</span>
                             )}
                          </div>
                          
                          {/* Custom Dashed progress bar */}
                          <div className="mt-3 flex gap-1 sm:gap-1.5 w-full h-1.5">
                             {Array.from({ length: season.airedCount }).map((_, epIdx) => {
                                const isWatched = season.watched.includes(epIdx + 1);
                                return (
                                  <div key={epIdx} className={cn("h-full flex-1 rounded-full", isWatched ? "bg-indigo-500 shadow-[0_0_5px_rgba(99,102,241,0.5)]" : "bg-[#1e2336]")} />
                                )
                             })}
                          </div>
                       </div>
                     </div>
                   );
                })}
              </div>
           </div>
